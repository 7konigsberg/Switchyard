4//! Pure commit-or-escalate policy for verification-gated routing.
5
6use super::{Branch, Capabilities, ToolErrorCount, select_branch};
7
8const POLICY_VERSION: &str = "1.0.0";
9const READOUT: f64 = 0.9;
10const DELIBERATION: f64 = 0.5;
11
12/// Tier selected for the terminal response.
13#[derive(Clone, Copy, Debug, Eq, PartialEq)]
14pub enum Route {
15    /// Serve the verified local attempt.
16    Local,
17    /// Escalate to the capable tier.
18    Cloud,
19}
20
21/// Definite verifier answer or unusable output.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23pub enum Tri {
24    /// Verifier affirmed.
25    Yes,
26    /// Verifier rejected.
27    No,
28    /// Verifier failed to return a usable answer.
29    Unknown,
30}
31
32/// Bounded evidence gathered for one local attempt.
33#[derive(Clone, Copy, Debug, Default, PartialEq)]
34pub struct Signals {
35    /// Cheap probability that the attempt is correct.
36    pub readout: Option<f64>,
37    /// Considered probability from the same verifier.
38    pub deliberation: Option<f64>,
39    /// Strict local evidence such as an observed successful test run.
40    pub strict_evidence: Option<Tri>,
41    /// Capable-tier confirmation when the coding dial is used.
42    pub cloud_judge: Option<Tri>,
43}
44
45/// Auditable result of applying the frozen policy.
46#[derive(Clone, Copy, Debug, PartialEq)]
47pub struct Decision {
48    /// Selected terminal tier.
49    pub route: Route,
50    /// Verification regime used.
51    pub branch: Branch,
52    /// Evidence supplied to the policy.
53    pub signals: Signals,
54    /// Frozen policy identity.
55    pub policy_version: &'static str,
56}
57
58/// Applies the frozen VGR policy. Missing or contradictory evidence escalates.
59pub fn decide_from_signals(caps: &Capabilities, signals: &Signals) -> Decision {
60    let branch = select_branch(caps);
61    let has_reported_error = matches!(
62        caps.tool_errors,
63        Some(ToolErrorCount::Host(count) | ToolErrorCount::Untrusted(count)) if count > 0
64    );
65    let commits = !has_reported_error
66        && match branch {
67            Branch::Coding => coding_commits(signals),
68            Branch::Chat => {
69                clears(signals.deliberation, DELIBERATION) || clears(signals.readout, 0.3)
70            }
71            Branch::Agentic => {
72                matches!(caps.tool_errors, Some(ToolErrorCount::Host(0)))
73                    && (clears(signals.deliberation, DELIBERATION) || clears(signals.readout, 0.2))
74            }
75            Branch::DefaultVerified => {
76                clears(signals.deliberation, DELIBERATION) || clears(signals.readout, 0.7)
77            }
78            Branch::Unknown => false,
79        };
80    Decision {
81        route: if commits { Route::Local } else { Route::Cloud },
82        branch,
83        signals: *signals,
84        policy_version: POLICY_VERSION,
85    }
86}
87
88fn coding_commits(signals: &Signals) -> bool {
89    let confident = clears(signals.readout, READOUT);
90    if confident && matches!(signals.cloud_judge, Some(Tri::No | Tri::Unknown)) {
91        return false;
92    }
93    let strict =
94        signals.strict_evidence == Some(Tri::Yes) && clears(signals.deliberation, DELIBERATION);
95    let dial = clears(signals.readout, 0.2) && signals.cloud_judge == Some(Tri::Yes);
96    strict || dial || confident
97}
98
99fn clears(score: Option<f64>, threshold: f64) -> bool {
100    score.is_some_and(|score| score >= threshold)
101}
