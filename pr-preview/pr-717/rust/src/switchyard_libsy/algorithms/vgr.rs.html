4//! Verification-gated routing capability derivation.
5//!
6//! Capabilities come only from router-produced task typing, the normalized
7//! request, and the local attempt. Missing or unrepresentable context selects
8//! [`Branch::Unknown`], so later routing fails closed.
9
10#![allow(dead_code)]
11
12use switchyard_protocol::Request;
13
14mod decide;
15mod text;
16
17#[cfg(test)]
18mod tests;
19
20/// Verification regime selected from trusted request and attempt evidence.
21#[derive(Clone, Copy, Debug, Eq, PartialEq)]
22pub enum Branch {
23    /// Code-producing work without an executable checker.
24    Coding,
25    /// Conversational or answer-seeking work.
26    Chat,
27    /// Tool-operating work.
28    Agentic,
29    /// Untyped work with a complete judged view.
30    DefaultVerified,
31    /// No complete evidence is available to verify.
32    Unknown,
33}
34
35/// Task type produced by the router's own typing call.
36#[derive(Clone, Copy, Debug, Eq, PartialEq)]
37pub enum TaskType {
38    /// Code-producing work.
39    Coding,
40    /// Tool-operating work.
41    Agentic,
42    /// Answer-seeking work.
43    Answer,
44    /// Conversational work.
45    Chat,
46}
47
48/// Tool-error count together with its trust provenance.
49#[derive(Clone, Copy, Debug, Eq, PartialEq)]
50pub(super) enum ToolErrorCount {
51    /// Count from host-owned execution evidence.
52    Host(i32),
53    /// Count derived from untrusted request history.
54    Untrusted(i32),
55}
56
57/// Complete input visible to the decision policy.
58#[derive(Clone, Debug, Default, PartialEq)]
59pub struct Capabilities {
60    /// Latest user instruction.
61    pub task_text: Option<String>,
62    /// Bounded, redacted request and local attempt.
63    pub transcript: Option<String>,
64    /// The request is coding work.
65    pub is_coding: bool,
66    /// The request is conversational or answer-seeking.
67    pub is_chat: bool,
68    /// The request is agentic.
69    pub is_agentic: bool,
70    /// No narrower regime was established.
71    pub default_verified: bool,
72    /// Locally generated attempt.
73    pub attempt: Option<String>,
74    /// Tool-error count supplied by the routing host or untrusted history.
75    pub tool_errors: Option<ToolErrorCount>,
76}
77
78/// Selects the strongest verification regime present in `caps`.
79pub fn select_branch(caps: &Capabilities) -> Branch {
80    if caps.is_coding {
81        Branch::Coding
82    } else if caps.is_chat {
83        Branch::Chat
84    } else if caps.is_agentic {
85        Branch::Agentic
86    } else if caps.default_verified {
87        Branch::DefaultVerified
88    } else {
89        Branch::Unknown
90    }
91}
92
93/// Derives policy capabilities without accepting client-declared routing flags.
94pub fn derive_capabilities(
95    request: &Request,
96    attempt: &str,
97    task_type: Option<TaskType>,
98    tool_errors: Option<ToolErrorCount>,
99) -> Capabilities {
100    let Some(view) = text::request_view(request, attempt) else {
101        return Capabilities::default();
102    };
103    let mut caps = Capabilities {
104        task_text: Some(view.task_text),
105        transcript: Some(view.transcript),
106        attempt: Some(attempt.to_string()),
107        tool_errors,
108        ..Default::default()
109    };
110
111    if text::observed_code_activity(attempt) || task_type == Some(TaskType::Coding) {
112        caps.is_coding = true;
113    } else if text::observed_tool_activity(attempt) || task_type == Some(TaskType::Agentic) {
114        caps.is_agentic = true;
115    } else if matches!(task_type, Some(TaskType::Answer | TaskType::Chat)) {
116        caps.is_chat = true;
117    } else {
118        caps.default_verified = true;
119    }
120    caps
121}
