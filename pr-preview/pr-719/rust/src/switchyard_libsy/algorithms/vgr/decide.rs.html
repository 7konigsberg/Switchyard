4//! The single decision function, and the readiness gates applied on top of it.
5//!
6//! [`decide_from_signals`] is where a branch, its evidence, and the host's own
7//! tool-error record combine into one commit-or-escalate verdict. It is pure: the
8//! same capabilities and signals always produce the same decision.
9//!
10//! # Two routes, not one
11//!
12//! A decision carries both what the rules concluded and what a deployment may act
13//! on. [`Decision::route`] is the rules' own verdict. [`Decision::effective_route`]
14//! applies the per-branch readiness gates, which force escalation on any branch
15//! whose safety precondition is unmet regardless of how strong its evidence was.
16//! Keeping both means a deployment can measure what the rules would have decided
17//! while serving only what it is ready to serve.
18

23/// Where a request is served.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25pub enum Route {
26    /// Commit the locally produced attempt.
27    Local,
28    /// Escalate to the capable tier.
29    Cloud,
30}
31
32/// Deployment state the readiness gates consult.
33///
34/// Separate from [`Capabilities`] because these are facts about the route's
35/// configuration, not about the request: no derivation over a request can
36/// establish them, and no request may influence them.
37#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
38pub struct Readiness {
39    /// An operator-configured sandboxed checker, with a pinned immutable-tests
40    /// manifest, was validated for this route.
41    pub checker_validated: bool,
42}
43
44/// The safety precondition a branch must meet before its local commits count.
45#[derive(Clone, Copy, Debug, Eq, PartialEq)]
46pub enum ReadinessGate {
47    /// The branch judges code, so it needs an operator-configured sandboxed
48    /// checker with a pinned test manifest. Coding without one is never
49    /// effectively local, whatever its evidence said.
50    SecureCheckerMissing,
51    /// The branch vetoes on tool errors, so the count must come from the
52    /// evidence source the routing host has chosen to trust.
53    ToolEvidenceNotHostAttested,
54}
55
56/// The full record of one decision.
57///
58/// Carries the evidence it was made from so a decision can be audited without
59/// re-running the verifiers that produced it.
60#[derive(Clone, Copy, Debug, PartialEq)]
61pub struct Decision {
62    /// What the rules concluded from the evidence.
63    pub route: Route,
64    /// What a deployment may act on, after the readiness gates.
65    pub effective_route: Route,
66    /// The gate that forced escalation, when one did.
67    pub readiness_gate: Option<ReadinessGate>,
68    /// The verification regime the capabilities selected.
69    pub branch: Branch,
70    /// The evidence the decision was made from.
71    pub signals: Signals,
72    /// The constant set the decision was made under.
73    pub policy_version: &'static str,
74}

76/// Resolves the tool-error evidence the veto acts on.
77///
78/// The host's own count is authoritative: a reported count can never override it,
79/// and a reporter that disagrees with it makes the evidence indeterminate rather
80/// than picking a winner. Provenance is read **only** when a count is actually
81/// present, so a host-provenance marker on an absent count never stands in for
82/// evidence that a trusted path produced something.
83///
84/// A clean count of untrusted provenance is discarded rather than believed. Such
85/// a count may still veto — a reported failure escalates whoever reported it —
86/// but it can never authorize a commit.
87fn resolve_tool_errors(caps: &Capabilities, sig: &Signals) -> ToolErrors {
88    let Some(reported) = caps.tool_errors else {
89        // A reported failure may still veto, but a reported clean count has no
90        // provenance and therefore cannot authorize an agentic commit.
91        return match sig.tool_errors {
92            ToolErrorSignal::Absent | ToolErrorSignal::NoCount => ToolErrors::NoInformation,
93            ToolErrorSignal::Indeterminate => ToolErrors::Indeterminate,
94            ToolErrorSignal::Count(0) => ToolErrors::NoInformation,
95            ToolErrorSignal::Count(count) => ToolErrors::Count(count),
96        };
97    };
98    let reported_count = reported.count();
99
100    let contradicts_host = match sig.tool_errors {
101        ToolErrorSignal::Absent => false,
102        ToolErrorSignal::NoCount | ToolErrorSignal::Indeterminate => true,
103        ToolErrorSignal::Count(count) => count != reported_count,
104    };
105    if contradicts_host {
106        return ToolErrors::Indeterminate;
107    }
108    if reported_count == 0 && !reported.is_host() {
109        return ToolErrors::NoInformation;
110    }
111    ToolErrors::Count(reported_count)
112}

114/// Whether host evidence describes the policy's bounded recovered-run shape.
115fn recovered_run_ok(caps: &Capabilities, tool_errors: ToolErrors, policy: &Policy) -> bool {
116    let config = policy.short_recovered_run;
117    if !config.enabled || !caps.tool_errors.is_some_and(ToolErrorCount::is_host) {
118        return false;
119    }
120    let ToolErrors::Count(errors) = tool_errors else {
121        return false;
122    };
123    let Some(results) = caps.tool_results else {
124        return false;
125    };
126    errors >= config.min_errors
127        && results >= 0
128        && results <= config.max_tool_results
129        && (!config.tail_clean || caps.tool_tail_clean == Some(true))
130}

132/// Whether the agentic branch can profitably spend local verification rungs.
133pub(super) fn agentic_can_gather(
134    caps: &Capabilities,
135    sig: &Signals,
136    policy: &Policy,
137) -> (bool, bool) {
138    let tool_errors = resolve_tool_errors(caps, sig);
139    let recovered = recovered_run_ok(caps, tool_errors, policy);
140    (
141        matches!(tool_errors, ToolErrors::Count(0)) || recovered,
142        recovered,
143    )
144}

146/// Decides whether to commit the local attempt or escalate.
147///
148/// Picks the branch the capabilities license, resolves the host tool-error veto,
149/// and applies that branch's rule. The veto is enforced here rather than inside
150/// the rules so that it binds every attempt-judging branch uniformly, and can
151/// never be sidestepped by a branch forgetting to consult it.
152///
153/// Two branches are exempt, for reasons specific to what their evidence is:
154/// checks, because a sandboxed test run is ground truth that supersedes
155/// trajectory noise, and answer, because its verifiers judge the output
156/// independently of how the attempt reached it.
157pub fn decide_from_signals(
158    caps: &Capabilities,
159    sig: &Signals,
160    policy: &Policy,
161    readiness: &Readiness,
162) -> Decision {
163    let branch = select_branch(caps);
164    let thr = &policy.thresholds;
165    let dial = policy.offload_dial.for_branch(branch);
166    let tool_errors = resolve_tool_errors(caps, sig);
167    // No evidence of errors does not veto; anything short of a clean, trusted
168    // count does.
169    let veto_ok = matches!(
170        tool_errors,
171        ToolErrors::NoInformation | ToolErrors::Count(0)
172    );
173
174    let recovered =
175        branch == Branch::AgenticVerified && recovered_run_ok(caps, tool_errors, policy);
176    let mut resolved_signals = *sig;
177    resolved_signals.recovered_run = recovered;
178    if recovered {
179        resolved_signals.tool_results = caps.tool_results;
180        resolved_signals.tool_tail_clean = caps.tool_tail_clean;
181    }
182
183    let commit = match branch {
184        Branch::Checks => rules::rule_checks(sig.tests_pass),
185        Branch::CodingNoChecks => {
186            rules::rule_coding_no_checks(
187                &resolved_signals,
188                thr,
189                dial,
190                policy.coding_dial_requires_judge,
191            ) && veto_ok
192        }
193        Branch::Answer => rules::rule_answer(sig, caps.structured_answer, policy),
194        Branch::Chat => rules::rule_chat(sig, thr, dial) && veto_ok,
195        // An operator's prior never outranks the host's own error count.
196        Branch::AgenticRecognized => {
197            rules::rule_agentic_recognized(caps.prior_local, thr) && veto_ok
198        }
199        // The veto is this branch's own rule, so it is not applied twice.
200        Branch::AgenticVerified => {
201            rules::rule_agentic_verified(tool_errors, &resolved_signals, thr, dial)
202        }
203        Branch::DefaultVerified => rules::rule_default_verified(sig, thr, dial) && veto_ok,
204        Branch::Unknown => false,
205    };
206
207    let route = if commit { Route::Local } else { Route::Cloud };
208    let (effective_route, readiness_gate) = readiness_effective(branch, route, caps, readiness);
209    Decision {
210        route,
211        effective_route,
212        readiness_gate,
213        branch,
214        signals: resolved_signals,
215        policy_version: policy.version,
216    }
217}

219/// Applies the per-branch readiness gates to a decided route.
220///
221/// A gate is a precondition on the *deployment*, not on the request: it asks
222/// whether the machinery a branch's local commits depend on actually exists yet.
223/// Until it does, that branch escalates even when its evidence was conclusive.
224/// Escalation is never gated, since there is nothing to make safe.
225pub fn readiness_effective(
226    branch: Branch,
227    route: Route,
228    caps: &Capabilities,
229    readiness: &Readiness,
230) -> (Route, Option<ReadinessGate>) {
231    if route != Route::Local {
232        return (Route::Cloud, None);
233    }
234    match branch {
235        // Both coding branches judge code, and neither is trustworthy without a
236        // validated checker: the branch that has one must have had it validated,
237        // and the branch that has none is never effectively local.
238        Branch::Checks | Branch::CodingNoChecks if !readiness.checker_validated => {
239            (Route::Cloud, Some(ReadinessGate::SecureCheckerMissing))
240        }
241        // Provenance is the whole gate here: these branches commit on the
242        // absence of tool errors, which only a host-trusted source can attest.
243        Branch::AgenticVerified | Branch::AgenticRecognized
244            if !matches!(caps.tool_errors, Some(ToolErrorCount::Host(_))) =>
245        {
246            (
247                Route::Cloud,
248                Some(ReadinessGate::ToolEvidenceNotHostAttested),
249            )
250        }
251        _ => (Route::Local, None),
252    }
