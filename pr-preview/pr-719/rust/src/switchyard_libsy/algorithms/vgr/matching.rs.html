4//! Typed, anchored agreement between a witness answer and an attempt's final
5//! message.
6//!
7//! Answering "do these two say the same thing?" mechanically is where a
8//! verification-gated router is easiest to fool, so this module is deliberately
9//! strict. Everything it accepts, a careful reader would call agreement; it
10//! rejects a great deal that a looser matcher would accept.
11//!
12//! Three properties do most of the work:
13//!
14//! - **Quantities are typed.** A unit is part of the value, so `$5`, `€5`, `5`
15//!   and `5%` are four different answers. Numbers compare by value, never by
16//!   prefix or substring.
17//! - **Matching is anchored.** A candidate must appear in the span the message
18//!   actually concludes with — an explicit answer marker, else the last
19//!   non-empty line. A number mentioned and then discarded earlier in the
20//!   reasoning is not the answer.
21//! - **Ambiguity is not agreement.** Same-line corrections resolve to the last
22//!   segment, and a span still carrying negation or contradiction language
23//!   cannot be resolved mechanically, so it yields [`Tri::Unknown`] rather than
24//!   a guess. Indeterminate never commits.
25

32/// Words too common or too syntactic to identify an answer on their own.
33///
34/// A candidate made only of these is a code fragment or filler that would match
35/// almost any message, not a claim about the task.
36static STOPWORDS: LazyLock<HashSet<&'static str>> = LazyLock::new(|| {
37    [
38        "the", "a", "an", "and", "or", "of", "to", "in", "on", "is", "are", "it", "yes", "no",
39        "none", "n/a", "na", "ok", "okay", "done", "true", "false", "if", "else", "for", "while",
40        "return", "report", "retry", "answer", "final", "result", "block", "task", "not",
41    ]
42    .into_iter()
43    .collect()
44});
45
46/// Words that turn a nearby candidate into a rejected one rather than an answer.
47static NEGATIONS: LazyLock<HashSet<&'static str>> = LazyLock::new(|| {
48    [
49        "not",
50        "no",
51        "never",
52        "isnt",
53        "wasnt",
54        "without",
55        "excluding",
56        "except",
57        "rather",
58        "instead",
59        "neither",
60        "nor",
61        "wrong",
62        "incorrect",
63    ]
64    .into_iter()
65    .collect()
66});

68/// How many tokens before a candidate are checked for negation.
69const NEGATION_WINDOW: usize = 3;

71/// A terse answer that is entirely a quantity, optionally signed or unit-bearing.
72static NUMBERISH: LazyLock<Option<Regex>> =
73    LazyLock::new(|| Regex::new(r"^[\s$€£+-]*[0-9][0-9,\s]*(?:\.[0-9]+)?\s*%?$").ok());
74
75/// A normalized token: a Unicode alphanumeric run, keeping an ASCII decimal tail.
76static TOKEN: LazyLock<Option<Regex>> =
77    LazyLock::new(|| Regex::new(r"[\p{L}\p{N}]+(?:\.[0-9]+)?").ok());
78
79/// A currency-prefixed quantity embedded in prose.
80static CURRENCY_QUANTITY: LazyLock<Option<Regex>> =
81    LazyLock::new(|| Regex::new(r"([$€£])\s*([0-9][0-9,]*(?:\.[0-9]+)?)").ok());
82
83/// A percent-suffixed quantity embedded in prose.
84static PERCENT_QUANTITY: LazyLock<Option<Regex>> =
85    LazyLock::new(|| Regex::new(r"([0-9][0-9,]*(?:\.[0-9]+)?)\s*%").ok());

87/// An explicit statement of the answer.
88static ANSWER_MARKER: LazyLock<Option<Regex>> = LazyLock::new(|| {
89    Regex::new(
90        r"(?i)\b(?:final\s+answer|revised\s+answer|answer|conclusion|result|correction|actually)\s*(?:is|:|=|-)\s*",
91    )
92    .ok()
93});
94
95/// A marker after which the operative answer restarts, used to split a span.
96static CORRECTION_SPLIT: LazyLock<Option<Regex>> = LazyLock::new(|| {
97    Regex::new(
98        r"(?i)(?:\b(?:actually|correction|revised(?:\s+answer)?|i\s+meant)\b[,:\s]*|(?:^|[,;])\s*(?:instead|rather)\s*[:,]\s*)",
99    )
100    .ok()
101});
102
103/// Contrasts join two viable values rather than introducing a correction.
104static INFIX_CONTRAST: LazyLock<Option<Regex>> =
105    LazyLock::new(|| Regex::new(r"(?i)\b(?:instead\s+of|rather\s+than)\b").ok());

107/// The unit a quantity carries. Part of the value, never discarded.
108#[derive(Clone, Copy, Debug, Eq, PartialEq)]
109enum Unit {
110    /// A bare number.
111    None,
112    /// A currency marker, kept as the character itself so currencies differ.
113    Currency(char),
114    /// A percentage.
115    Percent,
116}
117
118/// A parsed quantity: a magnitude and the unit that types it.
119#[derive(Clone, Debug, Eq, PartialEq)]
120struct Quantity {
121    value: String,
122    unit: Unit,
123}
124
125impl Quantity {
126    /// Whether two quantities are the same answer.
127    ///
128    /// Units must be identical, so `$5` never equals `5` or `€5`. Magnitudes
129    /// use a canonical decimal representation, so formatting differences are
130    /// ignored without losing precision.
131    fn agrees_with(&self, other: &Quantity) -> bool {
132        self.unit == other.unit && self.value == other.value
133    }
134}

136/// Canonicalizes a finite decimal without converting through a floating point type.
137fn canonical_decimal(s: &str) -> Option<String> {
138    let compact: String = s
139        .chars()
140        .filter(|c| !matches!(c, ',' | '+') && !c.is_whitespace())
141        .collect();
142    let (negative, unsigned) = match compact.strip_prefix('-') {
143        Some(unsigned) => (true, unsigned),
144        None => (false, compact.as_str()),
145    };
146    let mut parts = unsigned.split('.');
147    let integer = parts.next()?;
148    let fraction = parts.next();
149    if parts.next().is_some()
150        || integer.is_empty()
151        || !integer.chars().all(|c| c.is_ascii_digit())
152        || fraction
153            .is_some_and(|digits| digits.is_empty() || !digits.chars().all(|c| c.is_ascii_digit()))
154    {
155        return None;
156    }
157
158    let integer = match integer.trim_start_matches('0') {
159        "" => "0",
160        digits => digits,
161    };
162    let fraction = match fraction {
163        Some(digits) => digits.trim_end_matches('0'),
164        None => "",
165    };
166    let is_zero = integer == "0" && fraction.is_empty();
167    let sign = if negative && !is_zero { "-" } else { "" };
168    if fraction.is_empty() {
169        Some(format!("{sign}{integer}"))
170    } else {
171        Some(format!("{sign}{integer}.{fraction}"))
172    }
173}

175/// Parses a terse answer that is entirely a quantity.
176///
177/// Returns `None` for anything with surrounding prose. A string carrying two
178/// different currency markers, or both a currency and a percent, is not a
179/// coherent quantity and is rejected.
180fn parse_quantity(s: &str) -> Option<Quantity> {
181    let t = s.trim();
182    if t.is_empty()
183        || !NUMBERISH
184            .as_ref()
185            .is_some_and(|pattern| pattern.is_match(t))
186    {
187        return None;
188    }
189    // Which currency symbols appear at all, not how many times: a repeated
190    // marker is one currency, whereas two different markers are incoherent.
191    let mut present = ['$', '€', '£'].into_iter().filter(|c| t.contains(*c));
192    let currency = present.next();
193    if present.next().is_some() {
194        return None;
195    }
196    let unit = match (currency, t.contains('%')) {
197        (Some(_), true) => return None,
198        (Some(c), false) => Unit::Currency(c),
199        (None, true) => Unit::Percent,
200        (None, false) => Unit::None,
201    };
202    let digits: String = t
203        .chars()
204        .filter(|c| !matches!(c, ',' | '$' | '€' | '£' | '%' | '+') && !c.is_whitespace())
205        .collect();
206    canonical_decimal(&digits).map(|value| Quantity { value, unit })
207}

209/// The magnitude of a terse numeric answer, ignoring its unit.
210///
211/// Unit-blind, so it is used for deciding whether something *is* a number, never
212/// for deciding whether two numbers agree.
213fn parse_number(s: &str) -> Option<String> {
214    parse_quantity(s).map(|q| q.value)
215}

217/// Fuses unit markers onto their numbers before tokenization.
218///
219/// Without this, `$5 million` and `€5 million` tokenize identically and would
220/// agree. Fusing makes the unit part of the token, so unit-bearing and bare
221/// quantities can never match each other.
222fn fuse_units(s: &str) -> String {
223    let mut fused = s.to_string();
224    if let Some(pattern) = CURRENCY_QUANTITY.as_ref() {
225        fused = pattern
226            .replace_all(&fused, |caps: &regex::Captures| {
227                let marker = caps
228                    .get(1)
229                    .and_then(|capture| capture.as_str().chars().next())
230                    .map_or(0, u32::from);
231                let magnitude = match caps
232                    .get(2)
233                    .and_then(|capture| canonical_decimal(capture.as_str()))
234                {
235                    Some(value) => value,
236                    None => {
237                        return caps
238                            .get(0)
239                            .map_or("", |capture| capture.as_str())
240                            .to_string();
241                    }
242                };
243                format!(" cur{marker}u{magnitude} ")
244            })
245            .into_owned();
246    }
247    if let Some(pattern) = PERCENT_QUANTITY.as_ref() {
248        fused = pattern
249            .replace_all(&fused, |caps: &regex::Captures| {
250                let magnitude = match caps
251                    .get(1)
252                    .and_then(|capture| canonical_decimal(capture.as_str()))
253                {
254                    Some(value) => value,
255                    None => {
256                        return caps
257                            .get(0)
258                            .map_or("", |capture| capture.as_str())
259                            .to_string();
260                    }
261                };
262                format!(" pct{magnitude} ")
263            })
264            .into_owned();
265    }
266    fused
267}

269/// Normalizes text to the token sequence that matching compares.
270fn norm_tokens(s: &str) -> Vec<String> {
271    let fused: String = fuse_units(s)
272        .chars()
273        // Apostrophes are joiners inside words. Dropping every common variant
274        // makes `isn't` and `isn’t` the same negation token.
275        .filter(|c| !matches!(c, '\'' | '\u{2018}' | '\u{2019}' | '\u{02bc}' | '\u{ff07}'))
276        .flat_map(char::to_lowercase)
277        .collect();
278    match TOKEN.as_ref() {
279        Some(pattern) => pattern
280            .find_iter(&fused)
281            .map(|m| m.as_str().to_string())
282            .collect(),
283        None => Vec::new(),
284    }
285}

287/// Whether a candidate could plausibly identify an answer at all.
288///
289/// Guards against fragments that would match almost any message — a stopword, a
290/// scrap of punctuation-heavy code, a two-character sliver. Standalone numbers
291/// are always candidates; everything else must carry at least one substantial,
292/// non-stopword token and be mostly alphanumeric.
293fn informative(s: &str) -> bool {
294    let s = s.trim();
295    if parse_number(s).is_some() {
296        return true;
297    }
298    let length = s.chars().count();
299    if length < 3 {
300        return false;
301    }
302    let alphanumeric = s.chars().filter(|c| c.is_alphanumeric()).count();
303    if (alphanumeric as f64) / (length.max(1) as f64) < 0.6 {
304        return false;
305    }
306    norm_tokens(s)
307        .iter()
308        .any(|t| t.chars().count() >= 3 && !STOPWORDS.contains(t.as_str()))
309}

311/// Whether two terse answers agree.
312///
313/// Typed quantity equality when both sides are quantities, exact normalized
314/// token-sequence equality otherwise. A quantity never agrees with prose, and
315/// `Paris` never agrees with `Paris, Texas`.
316pub fn match_answers(a: &str, b: &str) -> bool {
317    match (parse_quantity(a), parse_quantity(b)) {
318        (Some(qa), Some(qb)) => qa.agrees_with(&qb),
319        // One side is a bare quantity and the other is not: different kinds of
320        // answer, so they cannot agree.
321        (Some(_), None) | (None, Some(_)) => false,
322        (None, None) => {
323            let ta = norm_tokens(a);
324            !ta.is_empty() && ta == norm_tokens(b)
325        }
326    }
327}

329/// The span a longform message actually concludes with.
330///
331/// The last explicit answer marker wins; failing that, the last non-empty line.
332/// A mention anywhere else in the text is reasoning, not a conclusion.
333fn extract_final_answer(message: &str) -> String {
334    if let Some(marker) = ANSWER_MARKER.as_ref()
335        && let Some(last) = marker.find_iter(message).last()
336    {
337        return message[last.end()..].trim().to_string();
338    }
339    message
340        .lines()
341        .map(str::trim)
342        .rfind(|line| !line.is_empty())
343        .map_or_else(String::new, str::to_string)
344}

346/// Whether any of the few tokens before position `index` negates what follows.
347fn negated_at(tokens: &[String], index: usize) -> bool {
348    tokens[index.saturating_sub(NEGATION_WINDOW)..index]
349        .iter()
350        .any(|t| NEGATIONS.contains(t.as_str()))
351}

353/// Reduces a span to its operative segment, reporting whether it stays ambiguous.
354///
355/// A same-line correction is structural, so `42, actually 17` resolves to `17`.
356/// A segment that still carries negation or contradiction language cannot be
357/// resolved mechanically at all — `X is not correct; Canada is` names two
358/// countries and endorses neither in a form this can read — so it is ambiguous,
359/// and ambiguity never becomes agreement.
360fn resolve_span(span: &str) -> (String, bool) {
361    let operative = match CORRECTION_SPLIT.as_ref().and_then(|pattern| {
362        pattern
363            .split(span)
364            .filter(|part| !part.trim().is_empty())
365            .last()
366    }) {
367        Some(last) => last.trim().to_string(),
368        None => span.trim().to_string(),
369    };
370    let ambiguous = INFIX_CONTRAST
371        .as_ref()
372        .is_some_and(|pattern| pattern.is_match(&operative))
373        || norm_tokens(&operative)
374            .iter()
375            .any(|t| NEGATIONS.contains(t.as_str()));
376    (operative, ambiguous)
377}

379/// A quantity found inside a span, with the byte offset it starts at.
380struct SpanQuantity {
381    quantity: Quantity,
382    start: usize,
383}
384
385/// Whether a character blocks a quantity from starting right after it.
386fn blocks_before(c: char) -> bool {
387    c.is_alphanumeric() || matches!(c, '_' | '.' | '€' | '£' | '$')
388}
389
390/// Whether a character blocks a quantity from ending right before it.
391fn blocks_after(c: char) -> bool {
392    c.is_alphanumeric() || matches!(c, '_' | '.' | '%')
393}

395/// Finds the quantities in a span that stand as values in their own right.
396///
397/// A quantity must be delimited: it may not run into a surrounding word,
398/// decimal, or unit marker. Without that, `42` would match inside `420`.
399///
400/// The reference expresses the delimiters as regex lookaround, which the `regex`
401/// crate does not support. Filtering whole regex matches after the fact is not
402/// equivalent, because a backtracking engine retries a *shorter* extent when the
403/// trailing delimiter fails — in `=5,,million`, `5,,` is rejected but `5,` is a
404/// valid quantity, and only backtracking finds it. This scans the span directly
405/// and reproduces that retry order: currency marker, then digit run, then
406/// decimal tail, then percent, each tried longest-first.
407fn span_quantities(span: &str) -> Vec<SpanQuantity> {
408    let chars: Vec<char> = span.chars().collect();
409    let at = |i: usize| chars.get(i).copied();
410    let mut found = Vec::new();
411    let mut start = 0;
412
413    while start < chars.len() {
414        if at(start.wrapping_sub(1)).is_some_and(|c| start > 0 && blocks_before(c)) {
415            start += 1;
416            continue;
417        }
418        match scan_quantity(&chars, start) {
419            Some((end, quantity)) => {
420                found.push(SpanQuantity {
421                    quantity,
422                    // Byte offset, for slicing the prefix the caller tokenizes.
423                    start: chars[..start].iter().map(|c| c.len_utf8()).sum(),
424                });
425                start = end.max(start + 1);
426            }
427            None => start += 1,
428        }
429    }
430    found
431}

433/// Scans one quantity at `start`, trying each quantifier longest-first.
434///
435/// Returns the end index and the parsed quantity for the first extent whose
436/// trailing delimiter holds, mirroring how a backtracking engine resolves the
437/// reference's pattern.
438fn scan_quantity(chars: &[char], start: usize) -> Option<(usize, Quantity)> {
439    let at = |i: usize| chars.get(i).copied();
440    let digit = |i: usize| at(i).is_some_and(|c| c.is_ascii_digit());
441
442    // `[$€£]?` is greedy: prefer consuming the marker, then try without it.
443    let has_marker = at(start).is_some_and(|c| matches!(c, '$' | '€' | '£'));
444    for marker_len in if has_marker {
445        [1, 0].as_slice()
446    } else {
447        [0].as_slice()
448    } {
449        let number_start = start + marker_len;
450        if !digit(number_start) {
451            continue;
452        }
453        // `[\d,]*` after the leading digit, longest-first.
454        let mut run = 0;
455        while at(number_start + 1 + run).is_some_and(|c| c.is_ascii_digit() || c == ',') {
456            run += 1;
457        }
458        for run_len in (0..=run).rev() {
459            let after_run = number_start + 1 + run_len;
460            // `(?:\.\d+)?` is greedy, and its `\d+` backtracks too.
461            let mut tail = 0;
462            if at(after_run) == Some('.') {
463                while digit(after_run + 1 + tail) {
464                    tail += 1;
465                }
466            }
467            let ends = (1..=tail)
468                .rev()
469                .map(|t| after_run + 1 + t)
470                .chain(std::iter::once(after_run));
471            for number_end in ends {
472                // `(%?)` is greedy, then the trailing delimiter must hold.
473                let with_percent = [(number_end + 1, true), (number_end, false)];
474                let without = [(number_end, false)];
475                let percent_options: &[(usize, bool)] = if at(number_end) == Some('%') {
476                    &with_percent
477                } else {
478                    &without
479                };
480                for (end, is_percent) in percent_options {
481                    if at(*end).is_some_and(blocks_after) {
482                        continue;
483                    }
484                    let text: String = chars[number_start..number_end].iter().collect();
485                    let Some(value) = parse_number(&text) else {
486                        continue;
487                    };
488                    let unit = match (*marker_len, *is_percent) {
489                        (1, _) => Unit::Currency(at(start)?),
490                        (_, true) => Unit::Percent,
491                        _ => Unit::None,
492                    };
493                    return Some((*end, Quantity { value, unit }));
494                }
495            }
496        }
497    }
498    None
499}

501/// Whether the answer appears in the span as the span's own claim.
502///
503/// A quantity answer matches a typed quantity of equal value and unit; a text
504/// answer matches the whole span, or a contiguous run of tokens within it. A
505/// single short token is not enough to match a longer span, since a common word
506/// would land anywhere. Any match preceded by negation is a rejected candidate,
507/// not an answer.
508fn match_in_span(answer: &str, span: &str) -> Tri {
509    let answer_tokens = norm_tokens(answer);
510
511    if let Some(target) = parse_quantity(answer) {
512        let mut candidates = span_quantities(span).into_iter().filter(|found| {
513            let prefix = norm_tokens(&span[..found.start]);
514            let length = prefix.len();
515            !negated_at(&prefix, length)
516        });
517        let candidate = match candidates.next() {
518            Some(found) => found,
519            None => return Tri::No,
520        };
521        if candidates.next().is_some() {
522            return Tri::Unknown;
523        }
524        return if candidate.quantity.agrees_with(&target) {
525            Tri::Yes
526        } else {
527            Tri::No
528        };
529    }
530
531    let span_tokens = norm_tokens(span);
532    if span_tokens == answer_tokens {
533        return Tri::Yes;
534    }
535    // A lone short token would match incidentally, so a run has to be either
536    // several tokens long or one substantial one.
537    let distinctive = answer_tokens.len() >= 2
538        || answer_tokens
539            .first()
540            .is_some_and(|t| t.chars().count() >= 5);
541    if !distinctive {
542        return Tri::No;
543    }
544    // A run longer than the span cannot occur in it, and `windows` requires a
545    // non-zero width; both are guarded here rather than relied on from above.
546    let k = answer_tokens.len();
547    if k == 0 || k > span_tokens.len() {
548        return Tri::No;
549    }
550    let mut matches = span_tokens
551        .windows(k)
552        .enumerate()
553        .filter(|(i, run)| *run == answer_tokens && !negated_at(&span_tokens, *i));
554    if matches.next().is_none() {
555        return Tri::No;
556    }
557    // Lists and coordinated phrases carry more than one plausible text value.
558    // Exact multi-token answers return above, while subset matches stay unknown.
559    if matches.next().is_some()
560        || span.chars().any(|c| matches!(c, ',' | ';'))
561        || span_tokens
562            .iter()
563            .any(|token| token == "and" || token == "or")
564    {
565        Tri::Unknown
566    } else {
567        Tri::Yes
568    }
569}

571/// Whether a witness answer agrees with an attempt's final message.
572///
573/// [`Tri::Unknown`] means the message's concluding span carries unresolved
574/// negation or contradiction: the answer branch treats that as no agreement and
575/// falls through to its verifier rungs, rather than guessing.
576pub fn match_answer_verdict(answer: &str, message: &str) -> Tri {
577    let answer = answer.trim();
578    if !informative(answer) || message.trim().is_empty() {
579        return Tri::No;
580    }
581    let (operative, ambiguous) = resolve_span(&extract_final_answer(message));
582    if ambiguous {
583        return Tri::Unknown;
584    }
585    match_in_span(answer, &operative)
586}
