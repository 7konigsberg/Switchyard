4//! The record of what one verification-gated turn did, and what it cost.
5//!
6//! [`Record`] is what an operator reads to answer "why did this turn route the
7//! way it did, and what did deciding it cost". It carries the decision itself,
8//! the per-rung timings and token counts behind it, and the flags that say how
9//! much of the evidence gathering actually completed.
10//!
11//! # Why this is separate from `Decision`
12//!
13//! [`Decision`](super::decide::Decision) is the output of a pure function: the
14//! same capabilities and signals always produce the same one. Timings, token
15//! counts and transport failures are none of those things — folding them in
16//! would make the decision core's output depend on when it ran. So the pure
17//! record stays pure and this one wraps it.
18//!
19//! # What may appear here
20//!
21//! Everything on this type is safe to log. Nothing derived from request or
22//! response *content* is recorded: errors are reduced to their class by
23//! [`safe_error_summary`](crate::algorithms::util::robustness::safe_error_summary)
24//! before they arrive, stage names are static labels, and the task type is a
25//! closed enum. There is deliberately no field for the attempt, the judged view,
26//! a verifier's reply, or an upstream error body.

40/// Identifies the prompt set the verifiers were called with.
41///
42/// A decision can only be compared against another decision made by the same
43/// questions, so the questions are versioned alongside the policy constants. The
44/// reference derives this by hashing its own source text, which has no Rust
45/// analogue; a hand-maintained version is the honest equivalent — it must be
46/// bumped when a prompt in [`rungs`](super::rungs) changes.
47pub(super) const PROMPT_VERSION: &str = "3";
48
49/// Which rung a timing belongs to.
50///
51/// A static label rather than a string so a stage name can never carry
52/// request-derived text into a log line.
53#[derive(Clone, Copy, Debug, Eq, PartialEq)]
54pub(super) enum Stage {
55    /// Producing the local attempt being judged.
56    Attempt,
57    /// Typing the request so derivation can select a regime.
58    Typing,
59    /// Running the sandboxed checker.
60    Checker,
61    /// Producing an independent witness answer for the agreement rung.
62    Witness,
63    /// The cheap probability-scored readout.
64    Readout,
65    /// The deliberating readout.
66    Deliberation,
67    /// A cloud confirmation call.
68    CloudJudge,
69    /// Judging a proposed tool-bearing assistant turn in flight.
70    TurnVerification,
71}
72
73impl Stage {
74    /// The label this stage is recorded under.
75    pub(super) fn label(self) -> &'static str {
76        match self {
77            Stage::Attempt => "attempt",
78            Stage::Typing => "typing",
79            Stage::Checker => "checker",
80            Stage::Witness => "witness",
81            Stage::Readout => "readout",
82            Stage::Deliberation => "deliberation",
83            Stage::CloudJudge => "cloud_judge",
84            Stage::TurnVerification => "turn_verification",
85        }
86    }
87}

89/// Why a turn produced no usable evidence from a rung.
90///
91/// Distinguishes the ways a rung can fail to answer, which the decision rules
92/// deliberately collapse into "indeterminate" but an operator needs apart: a
93/// verifier that was never configured is a deployment gap, one that timed out is
94/// a capacity problem, and one that hedged is a prompt problem.
95#[derive(Clone, Copy, Debug, Eq, PartialEq)]
96pub(super) enum Unknown {
97    /// The rung ran out of decision budget before it could start.
98    DeadlineExhausted,
99    /// The call was made and did not return in the remaining budget.
100    TimedOut,
101    /// The call failed. The class is recorded separately in `errors`.
102    CallFailed,
103    /// The verifier replied, but not with a bare verdict.
104    Unparsable,
105}
106
107impl Unknown {
108    /// The label this reason is recorded under.
109    fn label(self) -> &'static str {
110        match self {
111            Unknown::DeadlineExhausted => "deadline_exhausted",
112            Unknown::TimedOut => "timed_out",
113            Unknown::CallFailed => "call_failed",
114            Unknown::Unparsable => "unparsable",
115        }
116    }
117}

119/// What one verification-gated turn decided, and what deciding it cost.
120#[derive(Clone, Debug, Default)]
121pub(super) struct Record {
122    /// The decision, once one was made.
123    ///
124    /// `None` when the turn short-circuited before deciding — the kill switch,
125    /// the open circuit breaker, or a local tier that produced no attempt.
126    pub(super) decision: Option<Decision>,
127    /// The route actually served, which the serving mode may hold back.
128    pub(super) served_route: Option<Route>,
129    /// Why the turn short-circuited, when it did.
130    pub(super) short_circuit: Option<&'static str>,
131    /// Wall-clock time from the start of the turn to the decision.
132    pub(super) elapsed: Duration,
133    /// Per-rung timings, in the order the rungs ran.
134    pub(super) stages: Vec<(Stage, Duration)>,
135    /// Reasons rungs returned no usable evidence, by stage.
136    pub(super) unknowns: Vec<(Stage, Unknown)>,
137    /// Redacted failure classes. Never carries a message body.
138    pub(super) errors: Vec<String>,
139    /// Tokens billed to the local tier, across the attempt and the local rungs.
140    pub(super) local_tokens: u64,
141    /// Tokens billed to the cloud tier, across the confirmation rungs.
142    pub(super) cloud_tokens: u64,
143    /// Whether any rung was skipped or cut short because the budget ran out.
144    pub(super) deadline_exhausted: bool,
145    /// Whether the request carried content the router cannot faithfully judge.
146    pub(super) unsupported_content: bool,
147    /// Spans the baseline scrub replaced before any text was sent to a verifier.
148    pub(super) redaction_events: u32,
149    /// The router's own typing of the request, when it typed it.
150    pub(super) task_type: Option<TaskType>,
151    /// Provenance of the tool-error count the veto acted on, when there was one.
152    pub(super) tool_errors_source: Option<&'static str>,
153    /// Stable public identity of the checker contract used for this decision.
154    pub(super) checker_manifest: Option<String>,
155    /// The prompt set the verifiers were called with.
156    pub(super) prompt_version: &'static str,
157}
158
159impl Record {
160    /// A record for a turn that is starting now.
161    pub(super) fn new() -> Self {
162        Self {
163            prompt_version: PROMPT_VERSION,
164            ..Self::default()
165        }
166    }
167
168    /// Records how long a rung took.
169    pub(super) fn stage(&mut self, stage: Stage, elapsed: Duration) {
170        self.stages.push((stage, elapsed));
171    }
172
173    /// Carries the bounded decision labels to the terminal response for durable logging.
174    pub(super) fn annotate(&self, request: &mut Request) {
175        let extra = request
176            .metadata
177            .get_or_insert_default()
178            .extra_metadata
179            .get_or_insert_default();
180        for (key, value) in [
181            (
182                "switchyard.vgr.predicted",
183                route_label(self.decision.map(|d| d.route)),
184            ),
185            (
186                "switchyard.vgr.effective",
187                route_label(self.decision.map(|d| d.effective_route)),
188            ),
189            ("switchyard.vgr.served", route_label(self.served_route)),
190            (
191                "switchyard.vgr.branch",
192                branch_label(self.decision.map(|d| d.branch)),
193            ),
194            (
195                "switchyard.vgr.readiness_gate",
196                readiness_gate_label(self.decision.and_then(|d| d.readiness_gate)),
197            ),
198            (
199                "switchyard.vgr.short_circuit",
200                self.short_circuit.unwrap_or("none"),
201            ),
202        ] {
203            extra.insert(key.to_string(), value.to_string());
204        }
205    }
206
207    /// Records that a rung produced no usable evidence, and why.
208    ///
209    /// Budget-related reasons also set [`Record::deadline_exhausted`], so a
210    /// reader does not have to scan the list to learn the turn was cut short.
211    pub(super) fn unknown(&mut self, stage: Stage, reason: Unknown) {
212        if matches!(reason, Unknown::DeadlineExhausted | Unknown::TimedOut) {
213            self.deadline_exhausted = true;
214        }
215        self.unknowns.push((stage, reason));
216    }
217
218    /// Records a failure by class, discarding anything it carried.
219    pub(super) fn error(&mut self, stage: Stage, error: &crate::LibsyError) {
220        let summary = crate::algorithms::util::robustness::safe_error_summary(error);
221        self.errors.push(format!("{}: {summary}", stage.label()));
222    }
223
224    /// Emits the record as one structured event.
225    ///
226    /// Sequence fields are rendered rather than nested because the tracing field
227    /// set is flat; every rendered value is a static label or a number.
228    pub(super) fn emit(&self) {
229        meter().u64_counter(VGR_DECISIONS_METRIC).build().add(
230            1,
231            &[
232                KeyValue::new("predicted", route_label(self.decision.map(|d| d.route))),
233                KeyValue::new(
234                    "effective",
235                    route_label(self.decision.map(|d| d.effective_route)),
236                ),
237                KeyValue::new("served", route_label(self.served_route)),
238                KeyValue::new("branch", branch_label(self.decision.map(|d| d.branch))),
239                KeyValue::new(
240                    "readiness_gate",
241                    readiness_gate_label(self.decision.and_then(|d| d.readiness_gate)),
242                ),
243                KeyValue::new("short_circuit", self.short_circuit.unwrap_or("none")),
244            ],
245        );
246        let stages = self
247            .stages
248            .iter()
249            .map(|(stage, elapsed)| format!("{}={}", stage.label(), elapsed.as_millis()))
250            .collect::<Vec<_>>()
251            .join(",");
252        let unknowns = self
253            .unknowns
254            .iter()
255            .map(|(stage, reason)| format!("{}={}", stage.label(), reason.label()))
256            .collect::<Vec<_>>()
257            .join(",");
258        tracing::info!(
259            target: "libsy",
260            branch = ?self.decision.map(|decision| decision.branch),
261            route = ?self.decision.map(|decision| decision.route),
262            effective = ?self.decision.map(|decision| decision.effective_route),
263            gate = ?self.decision.and_then(|decision| decision.readiness_gate),
264            served = ?self.served_route,
265            short_circuit = self.short_circuit,
266            policy_version = self.decision.map(|decision| decision.policy_version),
267            prompt_version = self.prompt_version,
268            task_type = ?self.task_type,
269            tool_errors_source = ?self.tool_errors_source,
270            checker_manifest = self.checker_manifest.as_deref(),
271            elapsed_ms = self.elapsed.as_millis() as u64,
272            stages = %stages,
273            unknowns = %unknowns,
274            errors = ?self.errors,
275            local_tokens = self.local_tokens,
276            cloud_tokens = self.cloud_tokens,
277            deadline_exhausted = self.deadline_exhausted,
278            unsupported_content = self.unsupported_content,
279            redaction_events = self.redaction_events,
280            "vgr decision"
281        );
282    }
283}

285fn route_label(route: Option<Route>) -> &'static str {
286    match route {
287        Some(Route::Local) => "local",
288        Some(Route::Cloud) => "cloud",
289        None => "none",
290    }
291}
292
293fn branch_label(branch: Option<Branch>) -> &'static str {
294    match branch {
295        Some(Branch::Checks) => "checks",
296        Some(Branch::CodingNoChecks) => "coding_no_checks",
297        Some(Branch::Answer) => "answer",
298        Some(Branch::Chat) => "chat",
299        Some(Branch::AgenticRecognized) => "agentic_recognized",
300        Some(Branch::AgenticVerified) => "agentic_verified",
301        Some(Branch::DefaultVerified) => "default_verified",
302        Some(Branch::Unknown) => "unknown",
303        None => "none",
304    }
305}
306
307fn readiness_gate_label(gate: Option<ReadinessGate>) -> &'static str {
308    match gate {
309        Some(ReadinessGate::SecureCheckerMissing) => "secure_checker_missing",
310        Some(ReadinessGate::ToolEvidenceNotHostAttested) => "tool_evidence_not_host_attested",
311        None => "none",
312    }

315/// Labels a route retained by a processor that bypassed the VGR classifier.
316pub(super) fn annotate_retained_route(request: &mut Request, route: Route) {
317    let extra = request
318        .metadata
319        .get_or_insert_default()
320        .extra_metadata
321        .get_or_insert_default();
322    if extra.contains_key("switchyard.vgr.served") {
323        return;
324    }
325    let route = route_label(Some(route));
326    for (key, value) in [
327        ("switchyard.vgr.predicted", route),
328        ("switchyard.vgr.effective", route),
329        ("switchyard.vgr.served", route),
330        ("switchyard.vgr.branch", "affinity"),
331        ("switchyard.vgr.readiness_gate", "none"),
332        ("switchyard.vgr.short_circuit", "user_turn_affinity"),
333    ] {
334        extra.insert(key.to_string(), value.to_string());
335    }
336}

338/// Counts the spans the baseline scrub replaced in the judged material.
339///
340/// Counts placeholders in the rendered view rather than instrumenting the
341/// redactor: the view is what actually leaves for a verifier, so this measures
342/// the thing that matters and needs no plumbing through the pure render path.
343pub(super) fn count_redactions(judged: &str) -> u32 {
344    judged.matches("[REDACTED]").count() as u32
345}

360    #[derive(Default)]
361    struct FieldVisitor(BTreeMap<String, String>);
362
363    impl Visit for FieldVisitor {
364        fn record_str(&mut self, field: &Field, value: &str) {
365            self.0.insert(field.name().to_string(), value.to_string());
366        }
367
368        fn record_debug(&mut self, field: &Field, value: &dyn std::fmt::Debug) {
369            self.0
370                .insert(field.name().to_string(), format!("{value:?}"));
371        }
372    }
373
374    struct CaptureLayer(Arc<Mutex<Vec<BTreeMap<String, String>>>>);
375
376    impl<S> Layer<S> for CaptureLayer
377    where
378        S: Subscriber,
379    {
380        fn on_event(&self, event: &Event<'_>, _context: Context<'_, S>) {
381            let mut visitor = FieldVisitor::default();
382            event.record(&mut visitor);
383            self.0.lock().push(visitor.0);
384        }
385    }

387    #[test]
388    fn a_budget_reason_marks_the_turn_as_cut_short() {
389        // An operator reading `deadline_exhausted` must not have to scan the
390        // per-stage reasons to discover the turn ran out of budget.
391        let mut record = Record::new();
392        record.unknown(Stage::Readout, Unknown::TimedOut);
393        assert!(record.deadline_exhausted);
394
395        let mut other = Record::new();
396        other.unknown(Stage::Readout, Unknown::Unparsable);
397        assert!(!other.deadline_exhausted);
398    }
