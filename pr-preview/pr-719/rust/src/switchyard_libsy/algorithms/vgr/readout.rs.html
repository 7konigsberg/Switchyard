4//! The cheap readout: a few tokens of output, scored by token probability.
5//!
6//! The readout asks a verifier a yes/no question and reads the *probability* it
7//! assigned to "yes" rather than the word it happened to emit. That turns a
8//! four-token completion into a graded score, which is what makes it cheap
9//! enough to run before any of the costlier rungs.
10//!
11//! # Why this reads raw provider JSON
12//!
13//! The normalized response types carry no token probabilities: the decoder
14//! keeps `choices[].logprobs` out of [`AggLlmResponse`] entirely. What it does
15//! keep, under the default preservation policy, is the provider's original
16//! response body — so the probabilities are recoverable from there, and only
17//! from there. This module is the single place that dependency lives; if
18//! typed logprob support is added later, only [`p_yes`] changes.
19//!
20//! # Failure is silence, never noise
21//!
22//! A target that does not support logprobs, a streamed response, a provider
23//! whose body is shaped differently, a malformed payload — all yield `None`.
24//! The decision rules already treat an absent readout as evidence that never
25//! commits, so a deployment that cannot produce readouts degrades to its
26//! costlier rungs rather than misreading them.
27
28use serde_json::{Value, json};
29use switchyard_protocol::{AggLlmResponse, FormatId, Request, WireFormat};
30
31/// How many alternatives to request at the scored token position.
32///
33/// The verdict is one token, so a handful of alternatives is enough to find
34/// both "yes" and "no" among them.
35const TOP_ALTERNATIVES: u64 = 8;
36
37/// Completion budget for a readout call.
38///
39/// The prompt demands a single word; the few extra tokens absorb a leading
40/// space or a stray punctuation mark without paying for a sentence.
41pub(super) const MAX_OUTPUT_TOKENS: u64 = 4;
42
43/// Asks the provider for a direct verdict and its token probabilities.
44///
45/// Disabling reasoning is request-local: the serving model remains free to
46/// think on ordinary agent calls, while the four-token verifier budget reaches
47/// `yes` or `no`. Both logprob fields are required together.
48pub(super) fn request_logprobs(request: &mut Request) {
49    request.llm_request.reasoning.effort = Some("none".to_string());
50    let extensions = &mut request.llm_request.extensions.fields;
51    extensions.insert("logprobs".to_string(), json!(true));
52    extensions.insert("top_logprobs".to_string(), json!(TOP_ALTERNATIVES));
53}
54
55/// The probability the verifier assigned to "yes" at its first token.
56///
57/// Returns `None` whenever the answer cannot be established — see the module
58/// documentation on why that is deliberately indistinguishable from an
59/// unavailable verifier.
60pub(super) fn p_yes(response: &AggLlmResponse) -> Option<f64> {
61    let body = response
62        .preservation
63        .responses
64        .get(&FormatId::from(WireFormat::OpenAiChat))?;
65    let alternatives = body
66        .pointer("/choices/0/logprobs/content/0/top_logprobs")?
67        .as_array()?;
68
69    // The two verdicts are scored against each other rather than against the
70    // whole distribution, so probability mass on unrelated tokens — whitespace,
71    // punctuation, a stray capitalization — cannot dilute the verdict.
72    let mut yes = None;
73    let mut no = None;
74    for entry in alternatives {
75        let Some(probability) = entry.get("logprob").and_then(Value::as_f64) else {
76            continue;
77        };
78        let probability = probability.exp();
79        match normalize(entry.get("token")?.as_str()?) {
80            Verdict::Yes => *yes.get_or_insert(0.0) += probability,
81            Verdict::No => *no.get_or_insert(0.0) += probability,
82            Verdict::Other => {}
83        }
84    }
85
86    match (yes, no) {
87        // Both verdicts present: the score is yes against the pair.
88        (Some(yes), Some(no)) if yes + no > 0.0 => Some(yes / (yes + no)),
89        // Normalize over observed verdict mass, matching the reference: one
90        // observed verdict receives the whole binary mass.
91        (Some(_), None) => Some(1.0),
92        (None, Some(_)) => Some(0.0),
93        _ => None,
94    }
95}
96
97/// Which verdict, if either, a reported token stands for.
98enum Verdict {
99    Yes,
100    No,
101    Other,
102}
103
104/// Classifies a reported token, ignoring the framing a tokenizer adds.
105///
106/// Providers report the token as emitted, so the same verdict arrives variously
107/// as `yes`, `Yes`, or a space-prefixed form depending on the tokenizer.
108fn normalize(token: &str) -> Verdict {
109    let token = token
110        .trim()
111        .trim_matches(['"', '\'', '.', ','])
112        .to_lowercase();
113    match token.as_str() {
114        "yes" | "y" | "true" => Verdict::Yes,
115        "no" | "n" | "false" => Verdict::No,
116        _ => Verdict::Other,
117    }
118}

120#[cfg(test)]
121mod tests {
122    use super::*;
123    use switchyard_protocol::{LlmRequest, PreservationMetadata};
124
125    /// An aggregate carrying a preserved provider body with the given alternatives.
126    fn response_with(alternatives: Value) -> AggLlmResponse {
127        let mut preservation = PreservationMetadata::default();
128        preservation.responses.insert(
129            FormatId::from(WireFormat::OpenAiChat),
130            json!({"choices": [{"logprobs": {"content": [{"top_logprobs": alternatives}]}}]}),
131        );
132        AggLlmResponse {
133            preservation,
134            ..Default::default()
135        }
136    }
137
138    /// A reported alternative at the given natural probability.
139    fn alternative(token: &str, probability: f64) -> Value {
140        json!({"token": token, "logprob": probability.ln()})
141    }
142
143    #[test]
144    fn scores_yes_against_no_rather_than_the_whole_distribution() {
145        // Mass on unrelated tokens must not dilute the verdict.
146        let response = response_with(json!([
147            alternative("yes", 0.6),
148            alternative("no", 0.2),
149            alternative("\n", 0.2),
150        ]));
151        assert!(p_yes(&response).is_some_and(|score| (score - 0.75).abs() < 1e-9));
152    }
153
154    #[test]
155    fn reads_the_verdict_through_tokenizer_framing() {
156        // The same verdict arrives capitalized or space-prefixed by tokenizer.
157        for token in [" Yes", "YES", "yes.", "\"yes\""] {
158            let response = response_with(json!([alternative(token, 0.9), alternative("no", 0.1)]));
159            assert!(
160                p_yes(&response).is_some_and(|score| (score - 0.9).abs() < 1e-9),
161                "{token}"
162            );
163        }
164    }
165
166    #[test]
167    fn a_lone_verdict_receives_the_whole_observed_binary_mass() {
168        let only_yes = response_with(json!([alternative("yes", 0.97)]));
169        assert_eq!(p_yes(&only_yes), Some(1.0));
170        let only_no = response_with(json!([alternative("no", 0.95)]));
171        assert_eq!(p_yes(&only_no), Some(0.0));
172    }
173
174    #[test]
175    fn anything_unreadable_scores_nothing_rather_than_guessing() {
176        // No preserved body at all.
177        assert_eq!(p_yes(&AggLlmResponse::default()), None);
178        // A preserved body with no probabilities in it.
179        let mut preservation = PreservationMetadata::default();
180        preservation.responses.insert(
181            FormatId::from(WireFormat::OpenAiChat),
182            json!({"choices": [{"message": {"content": "yes"}}]}),
183        );
184        let bare = AggLlmResponse {
185            preservation,
186            ..Default::default()
187        };
188        assert_eq!(p_yes(&bare), None);
189        // Alternatives that contain neither verdict.
190        let neither = response_with(json!([alternative("maybe", 0.9)]));
191        assert_eq!(p_yes(&neither), None);
192    }
193
194    #[test]
195    fn readout_requests_direct_verdict_logprobs() {
196        // Thinking can consume the four-token budget before the verdict. Keep
197        // ordinary model calls unchanged while making this request answer directly.
198        let mut request = Request {
199            llm_request: LlmRequest::default(),
200            raw_request: None,
201            metadata: None,
202        };
203        request_logprobs(&mut request);
204        assert_eq!(
205            request.llm_request.reasoning.effort.as_deref(),
206            Some("none")
207        );
208        let fields = &request.llm_request.extensions.fields;
209        assert_eq!(fields.get("logprobs"), Some(&json!(true)));
210        assert_eq!(fields.get("top_logprobs"), Some(&json!(TOP_ALTERNATIVES)));
211    }
