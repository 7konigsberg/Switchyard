4//! The frozen decision constants: commit thresholds and the per-branch offload dial.
5//!
6//! Every number the decision rules consult lives here, so a policy revision is a
7//! change to this file rather than an edit scattered across the rules. The
8//! constants are frozen ahead of the measurements that justify them, which is why
9//! they are data rather than tuning knobs exposed to callers.
10
11use super::Branch;
12
13/// Commit thresholds shared by the decision rules.
14///
15/// `readout` is the cheap logprob score a verifier produces in a few tokens;
16/// `deliberation` is the same verifier's considered score, which costs more and
17/// is therefore held to a lower bar. `prior` applies only to an operator-declared
18/// agentic surface.
19#[derive(Clone, Copy, Debug, PartialEq)]
20pub struct Thresholds {
21    /// Bar the cheap readout must clear to commit on its own.
22    pub readout: f64,
23    /// Floor of the uncertain readout band, below which only local evidence commits.
24    pub readout_band_low: f64,
25    /// Bar the deliberating readout must clear.
26    pub deliberation: f64,
27    /// Bar an operator-declared prior must clear.
28    pub prior: f64,
29}
30
31/// The per-branch offload dial: an extra readout bar that licenses a commit.
32///
33/// Each entry adds a readout arm to its branch's rule. The coding arm also
34/// requires its grading judge to affirm. A `None` entry disables the arm for
35/// that branch, which reproduces the pre-dial policy exactly. The dial only ever
36/// adds commits; it can never turn a commit into an escalation, and every veto
37/// that binds a branch also binds its dial arm.
38///
39/// Lowering a dial admits more local commits, which raises the share of traffic
40/// served locally. The values are an empirical selection, not a derivation.
41#[derive(Clone, Copy, Debug, Default, PartialEq)]
42pub struct OffloadDial {
43    /// Bar for the coding-without-checks branch.
44    pub coding: Option<f64>,
45    /// Bar for the conversational branch.
46    pub chat: Option<f64>,
47    /// Bar for the typed-answer branch.
48    pub answer: Option<f64>,
49    /// Bar for the untyped default branch.
50    pub default_verified: Option<f64>,
51    /// Bar for the evidence-verified agentic branch.
52    pub agentic: Option<f64>,
53}
54
55impl OffloadDial {
56    /// The dial bar for a branch, or `None` when the branch has no dial arm.
57    ///
58    /// Branches absent from the dial — checks and the operator-prior agentic
59    /// branch — never gain a readout arm, because neither consults a readout.
60    pub fn for_branch(&self, branch: Branch) -> Option<f64> {
61        match branch {
62            Branch::CodingNoChecks => self.coding,
63            Branch::Chat => self.chat,
64            Branch::Answer => self.answer,
65            Branch::DefaultVerified => self.default_verified,
66            Branch::AgenticVerified => self.agentic,
67            Branch::Checks | Branch::AgenticRecognized | Branch::Unknown => None,
68        }
69    }
70}

72/// Shape that lets a short agentic run recover from earlier tool errors.
73#[derive(Clone, Copy, Debug, PartialEq)]
74pub struct ShortRecoveredRun {
75    /// Whether the recovery arm is part of this policy.
76    pub enabled: bool,
77    /// Least number of earlier tool errors needed to call the run recovered.
78    pub min_errors: i32,
79    /// Most tool results a recovered run may contain.
80    pub max_tool_results: i32,
81    /// Whether the final tool result must be clean.
82    pub tail_clean: bool,
83}
84
85/// Frozen controls for judging proposed tool-bearing assistant turns.
86#[derive(Clone, Copy, Debug, PartialEq)]
87pub struct TurnVerification {
88    /// Whether proposed tool-bearing turns are judged.
89    pub enabled: bool,
90    /// Consecutive escalation votes required to latch a session.
91    pub confirmations: u32,
92    /// Probability at or above which a judgment is an escalation vote.
93    pub escalate_at: f64,
94    /// Number of recent conversation messages retained.
95    pub recent_messages: usize,
96    /// Per-message character budget in the recent window.
97    pub message_chars: usize,
98    /// Character budget for the system anchor.
99    pub system_chars: usize,
100    /// Character budget for the opening user-task anchor.
101    pub first_user_chars: usize,
102    /// Backstop for the complete rendered trajectory.
103    pub max_chars: usize,
104}

106/// The complete constant set a decision is made under.
107///
108/// Carried on every [`Decision`](super::Decision) as `policy_version`, so a
109/// recorded decision can be replayed against the constants that produced it.
110#[derive(Clone, Copy, Debug, PartialEq)]
111pub struct Policy {
112    /// Identifier of this constant set, recorded on every decision.
113    pub version: &'static str,
114    /// Commit thresholds.
115    pub thresholds: Thresholds,
116    /// The per-branch offload dial.
117    pub offload_dial: OffloadDial,
118    /// Whether the typed-answer branch may commit on the universal transcript
119    /// judge alone.
120    ///
121    /// Disabled: local judges were measured unable to verify evidence-light
122    /// answers, so an answer commits on independent verification, on typed
123    /// agreement over an operator-declared structured surface, or at the dial
124    /// bar — never on the judge alone.
125    pub answer_judge_arms: bool,
126    /// Whether the coding dial arm requires a strict cloud-judge affirmation.
127    pub coding_dial_requires_judge: bool,
128    /// Host-attested short-run recovery from earlier tool errors.
129    pub short_recovered_run: ShortRecoveredRun,
130    /// In-flight verification of proposed tool-bearing turns.
131    pub turn_verification: TurnVerification,
132}
